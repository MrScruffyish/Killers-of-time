# The tests share one precompiled header (each test is a single file): built once in
# kot_test_pch and reused. GCC uses it only when the macros match, so every test has
# KOT_DATA_DIR and Qt Network; the client tests (Qt Widgets) compile without it.
if(KOT_PCH)
    add_library(kot_test_pch STATIC pch_dummy.cpp)
    target_link_libraries(kot_test_pch PRIVATE kot_core Qt6::Test Qt6::Network kot_warnings)
    target_compile_definitions(kot_test_pch PRIVATE KOT_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
    target_precompile_headers(kot_test_pch PRIVATE ${KOT_PCH_CORE} <QtTest>)
endif()

# kot_add_test(name [GUI]): GUI tests (Qt Widgets, run offscreen) link the client and
# compile without the shared precompiled header (their Qt macros differ).
function(kot_add_test name)
    add_executable(${name} ${name}.cpp)
    if("GUI" IN_LIST ARGN)
        target_link_libraries(${name} PRIVATE kot_client_lib kot_core Qt6::Test kot_warnings)
        target_compile_definitions(${name} PRIVATE KOT_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        return()
    endif()
    # Every test gets the data folder and Qt Network, so their flags match the shared header's.
    target_link_libraries(${name} PRIVATE kot_core Qt6::Test Qt6::Network kot_warnings)
    target_compile_definitions(${name} PRIVATE KOT_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
    if(KOT_PCH)
        target_precompile_headers(${name} REUSE_FROM kot_test_pch)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
