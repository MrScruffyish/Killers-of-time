function(kot_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE kot_core Qt6::Test kot_warnings)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

kot_add_test(tst_rng)
kot_add_test(tst_strongid)
kot_add_test(tst_unitstack)
kot_add_test(tst_jsonreader)
kot_add_test(tst_content)
target_compile_definitions(tst_content PRIVATE KOT_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
kot_add_test(tst_army)

target_compile_definitions(tst_herorules PRIVATE KOT_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
